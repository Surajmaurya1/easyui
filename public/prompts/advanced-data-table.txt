## Usage

Import path and example from component metadata:

```tsx
import { AdvancedDataTable, type ColumnDef } from "@/components/ui/advanced-data-table";

interface UserRecord {
  id: string;
  name: string;
  role: string;
  status: 'active' | 'pending';
}

const columns: ColumnDef<UserRecord>[] = [
  { id: 'name', header: 'User', accessorKey: 'name', sortable: true },
  { id: 'role', header: 'Role', accessorKey: 'role', sortable: true },
  { id: 'status', header: 'Status', accessorKey: 'status', filterable: true },
];

export function Demo() {
  const data: UserRecord[] = [
    { id: '1', name: 'Alex Chen', role: 'Engineer', status: 'active' },
    { id: '2', name: 'Elena Rostova', role: 'Designer', status: 'pending' },
  ];

  return (
    <AdvancedDataTable
      title="Team Members"
      data={data}
      columns={columns}
      onBulkDelete={(ids) => console.log('Delete:', ids)}
    />
  );
}
```

Declared props:
- `data` (T[]); default: [] — Array of data records to render in the table
- `columns` (ColumnDef<T>[]); default: [] — Column configuration defining headers, accessors, sorting, filtering, and custom cell renderers
- `title` (string); default: undefined — Optional table title displayed in the toolbar
- `searchPlaceholder` (string); default: 'Search records...' — Placeholder text for the global search input
- `defaultPageSize` (number); default: 10 — Initial number of records to show per page
- `renderSubComponent` ((row: T) => ReactNode); default: undefined — Render prop function for expandable row sub-details
- `onBulkDelete` ((selectedIds: string[]) => void); default: undefined — Callback fired when user executes bulk deletion
- `onBulkExport` ((selectedIds: string[]) => void); default: undefined — Callback fired when user clicks bulk export
- `isLoading` (boolean); default: false — Renders animated skeleton placeholders when true
- `error` (string | null); default: null — Renders an error state banner when present
- `className` (string); default: undefined — Custom Tailwind class names for styling overrides
