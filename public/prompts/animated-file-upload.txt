## Usage

Import path and example from component metadata:

```tsx
import { AnimatedFileUpload } from "@/components/ui/animated-file-upload";

export function Demo() {
  return (
    <div className="max-w-md mx-auto p-4">
      <AnimatedFileUpload
        multiple
        maxSize={10 * 1024 * 1024}
        accept="image/*,application/pdf"
        onFilesSelected={(files) => console.log('Selected:', files)}
        onUploadComplete={(file) => console.log('Uploaded:', file.name)}
      />
    </div>
  );
}
```

Declared props:
- `multiple` (boolean); default: true — Allow multiple files selection and upload
- `accept` (string | string[]); default: undefined — Accepted MIME types or file extensions (e.g. image/*, .pdf)
- `maxSize` (number); default: 26214400 (25MB) — Maximum file size in bytes
- `maxFiles` (number); default: 10 — Maximum number of concurrent files in list
- `dropTitle` (string); default: 'Drop files here' — Primary drop target heading
- `dropSubtitle` (string); default: 'or browse from your device' — Secondary call-to-action text
- `variant` ('standard' | 'compact'); default: 'standard' — Display density mode
- `disabled` (boolean); default: false — Disables all interactions and file picking
- `onFilesSelected` ((files: File[]) => void); default: undefined — Callback triggered when files are chosen
- `onUploadComplete` ((file: UploadFileItem) => void); default: undefined — Callback fired on successful upload completion
- `uploadHandler` ((file, onProgress) => Promise<void>); default: undefined — Custom async upload handler returning a promise
