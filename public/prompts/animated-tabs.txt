## Usage

Import path and example from component metadata:

```tsx
import { AnimatedTabs } from "@/components/ui/animated-tabs";

export function Demo() {
  const tabs = [
    { id: 'overview', label: 'Overview', content: <div>Metrics Overview</div> },
    { id: 'analytics', label: 'Analytics', content: <div>Traffic Charts</div> },
    { id: 'settings', label: 'Settings', content: <div>Preferences</div> },
  ];
  return <AnimatedTabs tabs={tabs} defaultTab="overview" />;
}
```

Declared props:
- `tabs` (TabItem[]); default: [] — Array of tabs with id, label, icon, content
- `defaultTab` (string); default: tabs[0].id — Initial active tab ID
- `onChange` ((id: string) => void); default: undefined — Tab change callback
