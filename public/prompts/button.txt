## Usage

Import path and example from component metadata:

```tsx
import { Button } from "@/components/ui/button";
import { Sparkles, ArrowRight } from "lucide-react";

export function Demo() {
  return (
    <div className="flex flex-wrap items-center gap-3">
      <Button variant="primary" leftIcon={<Sparkles className="w-4 h-4" />}>
        Get Started
      </Button>
      <Button variant="secondary">Secondary</Button>
      <Button variant="outline">Outline</Button>
      <Button variant="ghost">Ghost</Button>
      <Button variant="destructive">Destructive</Button>
    </div>
  );
}
```

Declared props:
- `variant` ('primary' | 'secondary' | 'outline' | 'ghost' | 'destructive' | 'success' | 'link' | 'gradient'); default: 'primary' — Visual presentation style
- `size` ('sm' | 'md' | 'lg' | 'icon'); default: 'md' — Dimensions and typography scale
- `isLoading` (boolean); default: false — Displays an animated spinner and disables user interaction
- `loadingText` (string); default: undefined — Optional text displayed alongside the loading spinner
- `leftIcon` (React.ReactNode); default: undefined — Icon element placed before children
- `rightIcon` (React.ReactNode); default: undefined — Icon element placed after children
- `fullWidth` (boolean); default: false — Stretches button to 100% width of parent container
- `disabled` (boolean); default: false — Prevents interaction and applies 30% disabled opacity
