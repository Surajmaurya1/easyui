## Usage

Import path and example from component metadata:

```tsx
import { Calendar, type DateRange } from "@/components/ui/calendar";

// single
const [date, setDate] = React.useState<Date | null>(null);
<Calendar value={date} onChange={setDate} />

// range
const [range, setRange] = React.useState<DateRange>({ from: null, to: null });
<Calendar mode="range" value={range} onChange={setRange} />

// limits
<Calendar minDate={new Date()} isDateDisabled={(d) => d.getDay() === 0} weekStartsOn={1} />
```

Declared props:
- `mode` ('single' | 'range'); default: 'single' — Selection mode
- `value` (Date | null | DateRange); default: undefined — Controlled selected date or range
- `defaultValue` (Date | null | DateRange); default: null — Initial selection (uncontrolled)
- `onChange` ((date: Date | null) => void | (range: DateRange) => void); default: undefined — Selection change callback
- `defaultMonth` (Date); default: undefined — Month shown initially
- `minDate` (Date); default: undefined — Earliest selectable date
- `maxDate` (Date); default: undefined — Latest selectable date
- `isDateDisabled` ((date: Date) => boolean); default: undefined — Predicate to disable specific dates
- `weekStartsOn` (0 | 1); default: 0 — First day of the week (0 = Sunday, 1 = Monday)
- `locale` (string); default: undefined — BCP 47 locale for formatting
- `className` (string); default: undefined — Additional class for the root
