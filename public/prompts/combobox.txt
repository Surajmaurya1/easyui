## Usage

Import path and example from component metadata:

```tsx
import { Combobox, type ComboboxOption } from "@/components/ui/combobox";

const frameworks: ComboboxOption[] = [
  { value: "next", label: "Next.js", description: "React framework" },
  { value: "remix", label: "Remix", description: "Web standards" },
  { value: "astro", label: "Astro", description: "Content sites", keywords: ["islands"] },
  { value: "vite", label: "Vite", disabled: true },
];

const [fw, setFw] = React.useState<string | null>("next");

<Combobox options={frameworks} value={fw} onValueChange={setFw} placeholder="Framework" />
```

Declared props:
- `options` (ComboboxOption[]); default: undefined — Options with value, label, description, icon, keywords, disabled
- `value` (string | null); default: undefined — Controlled selected value
- `defaultValue` (string | null); default: null — Initial value (uncontrolled)
- `onValueChange` ((value: string | null) => void); default: undefined — Selection change callback
- `placeholder` (string); default: 'Select…' — Trigger placeholder
- `searchPlaceholder` (string); default: 'Search…' — Search input placeholder
- `emptyText` (string); default: 'No results.' — Empty state message
- `align` ('start' | 'end'); default: 'start' — Popover alignment
- `disabled` (boolean); default: false — Disables the trigger
- `className` (string); default: undefined — Additional class for the wrapper (set width, e.g. w-72)
- `aria-label` (string); default: placeholder — Accessible name for the combobox
