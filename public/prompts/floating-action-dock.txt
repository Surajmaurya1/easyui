## Usage

Import path and example from component metadata:

```tsx
import { FloatingActionDock } from "@/components/ui/floating-action-dock";
import { Terminal, Code2, Sparkles, Settings } from "lucide-react";

export function Demo() {
  const items = [
    { id: 'terminal', label: 'Terminal', icon: <Terminal /> },
    { id: 'editor', label: 'Editor', icon: <Code2 /> },
    { id: 'ai', label: 'AI Assistant', icon: <Sparkles /> },
  ];
  return <FloatingActionDock items={items} activeId="terminal" />;
}
```

Declared props:
- `items` (DockItem[]); default: [] — Dock icons with labels, actions, and icons
- `activeId` (string); default: undefined — Current active item identifier
