## Usage

Import path and example from component metadata:

```tsx
import { Form, FormItem, FormLabel, FormControl, FormDescription, FormMessage, Input, Button } from "@/components/ui/form";
import { useState } from "react";

export function Demo() {
  const [email, setEmail] = useState("");
  const [error, setError] = useState("");

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!email.includes("@")) {
      setError("Please enter a valid email address.");
    } else {
      setError("");
    }
  };

  return (
    <Form onSubmit={handleSubmit} className="max-w-sm">
      <FormItem>
        <FormLabel required>Email Address</FormLabel>
        <FormControl>
          <Input
            type="email"
            placeholder="you@company.com"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            error={!!error}
          />
        </FormControl>
        <FormDescription>We will never share your email.</FormDescription>
        <FormMessage error={error} />
      </FormItem>
      <Button type="submit" variant="primary" fullWidth>
        Submit
      </Button>
    </Form>
  );
}
```

Declared props:
- `onSubmit` ((e: FormEvent) => void); default: undefined — Form submission handler
- `error` (string | boolean); default: undefined — Validation error text or boolean trigger
- `showPasswordToggle` (boolean); default: false — Enables eye icon toggle for password inputs
- `required` (boolean); default: false — Displays red asterisk and enforces requirement
- `leftIcon` (React.ReactNode); default: undefined — Leading icon inside input fields
- `rightIcon` (React.ReactNode); default: undefined — Trailing icon inside input fields
