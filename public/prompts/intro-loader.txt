## Usage

Import path and example from component metadata:

```tsx
import { IntroLoader } from "@/components/ui/intro-loader";
import { useState } from "react";

export function Demo() {
  const [loading, setLoading] = useState(true);

  if (loading) {
    return (
      <IntroLoader
        allowSkip={true}
        showProgress={true}
        onComplete={() => setLoading(false)}
      />
    );
  }

  return (
    <main className="min-h-screen bg-[#050505] text-[#FAFAFA] p-8">
      <h1>Welcome to the Application</h1>
    </main>
  );
}
```

Declared props:
- `greetings` ((string | GreetingItem)[]); default: ['Hello', 'Hola', 'Bonjour', ...] — Array of localized greeting strings or objects with text and lang to cycle through
- `fullScreen` (boolean); default: true — Whether to render as a fixed full-screen overlay or inline container
- `intervalMs` (number); default: 240 — Milliseconds per greeting cycle
- `showProgress` (boolean); default: true — Whether to display the Apple-style hairline progress bar
- `showLangBadge` (boolean); default: true — Whether to display country language badges
- `allowSkip` (boolean); default: true — Whether to enable ESC key and Skip button dismissals
- `speedMultiplier` (number); default: 1 — Speed multiplier for the animation sequence
- `onComplete` (() => void); default: undefined — Callback fired when welcome sequence completes
