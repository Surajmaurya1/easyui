## Usage

Import path and example from component metadata:

```tsx
import { Loader } from "@/components/ui/loader";

export function Demo() {
  return (
    <div className="flex items-center gap-6 p-8 bg-[#0A0A0A] rounded-2xl border border-[#222222]">
      <Loader variant="arc" size={32} />
      <Loader variant="dots" size={28} />
      <Loader variant="line" size={36} />
      <Loader variant="rings" size={32} />
    </div>
  );
}
```

Declared props:
- `size` (number); default: 32 — Diameter or width scale in pixels
- `variant` ('arc' | 'dots' | 'line' | 'rings'); default: 'arc' — Visual animation mode
- `label` (string); default: 'Loading...' — Accessible screen reader text label
- `reduceMotion` (boolean); default: false — Disables high-velocity continuous spins
- `color` (string); default: 'currentColor' — CSS color string for indicator strokes
- `className` (string); default: undefined — Additional CSS class names
