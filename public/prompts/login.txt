## Usage

Import path and example from component metadata:

```tsx
import { Login } from "@/components/ui/login";

export function Demo() {
  const handleLogin = async (data: { email: string; password: string; rememberMe: boolean }) => {
    console.log("Authenticating:", data);
  };

  return (
    <div className="py-8 flex justify-center">
      <Login
        onSubmit={handleLogin}
        onForgotPassword={() => alert("Redirect to forgot password")}
        onSignUpClick={() => alert("Redirect to sign up")}
      />
    </div>
  );
}
```

Declared props:
- `title` (string); default: 'Welcome back' — Primary card title text
- `description` (string); default: 'Sign in to access your EasyUI workspace' — Subtitle description below the title
- `logo` (React.ReactNode); default: <SparklesIcon /> — Brand badge or logo displayed at the top
- `error` (string | null); default: null — Server-side or authentication error banner message
- `isLoading` (boolean); default: false — Submitting state displaying loader on submit button
- `onSubmit` ((data: LoginFormData) => void); default: undefined — Form submission callback with email, password, rememberMe
- `onForgotPassword` (() => void); default: undefined — Callback when forgot password link is clicked
- `onSignUpClick` (() => void); default: undefined — Callback for secondary sign up switch action
- `showSocialLogins` (boolean); default: true — Toggles GitHub and Google SSO buttons
- `onSocialLogin` ((provider: 'github' | 'google' | 'apple') => void); default: undefined — Callback when social login button is pressed
