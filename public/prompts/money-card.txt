## Usage

Import path and example from component metadata:

```tsx
import { MoneyCard } from "@/components/ui/money-card";

export function Demo() {
  return (
    <div className="w-full py-12 flex justify-center">
      <MoneyCard
        balance={52002.5}
        onSend={async ({ contact, amount }) => {
          console.log(`Sent ${amount} to ${contact.name}`);
        }}
        onRequest={async ({ contact, amount }) => {
          console.log(`Requested ${amount} from ${contact.name}`);
        }}
      />
    </div>
  );
}
```

Declared props:
- `user` ({ name: string; avatar: string }); default: DEFAULT_USER — User profile with display name and avatar icon
- `balance` (number); default: 52002.5 — Headline account balance in major currency units
- `contacts` (Contact[]); default: DEFAULT_CONTACTS — List of quick-send contacts with avatars and tints
- `transactions` (Transaction[]); default: DEFAULT_TRANSACTIONS — List of recent transactions sorted newest first
- `onSend` ((req: TransferRequest) => Promise<void>); default: undefined — Async handler called upon submitting a send transfer
- `onRequest` ((req: TransferRequest) => Promise<void>); default: undefined — Async handler called upon submitting a request transfer
- `onViewAll` (() => void); default: undefined — Optional callback overriding default activity list opening
- `currency` (string); default: 'USD' — ISO currency symbol used for formatted monetary values
- `locale` (string); default: 'en-US' — Locale string for number and date formatting
- `paymentLink` (string); default: 'pay.example/ana' — Personal payment link displayed on receive and settings sheets
- `className` (string); default: '' — Optional CSS class overrides for the outer container
