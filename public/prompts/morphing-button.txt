## Usage

Import path and example from component metadata:

```tsx
import { MorphingButton } from "@/components/ui/morphing-button";

export function Demo() {
  const [status, setStatus] = useState<'idle' | 'loading' | 'success'>('idle');

  const handleClick = () => {
    setStatus('loading');
    setTimeout(() => {
      setStatus('success');
      setTimeout(() => setStatus('idle'), 2000);
    }, 1500);
  };

  return (
    <MorphingButton
      status={status}
      idleText="Deploy Project"
      loadingText="Building Edge..."
      successText="Deployed ✓"
      onClick={handleClick}
    />
  );
}
```

Declared props:
- `status` ('idle' | 'loading' | 'success' | 'error'); default: 'idle' — Current button lifecycle state
- `idleText` (string); default: 'Save Changes' — Label shown in default resting state
- `loadingText` (string); default: 'Saving...' — Label shown when operation is pending
- `successText` (string); default: 'Saved' — Label shown upon successful completion
- `errorText` (string); default: 'Failed' — Label shown when operation fails
- `variant` ('primary' | 'secondary' | 'danger' | 'ghost'); default: 'primary' — Visual surface styling preset
