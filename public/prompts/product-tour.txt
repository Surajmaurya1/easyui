## Usage

Import path and example from component metadata:

```tsx
import { useState } from "react";
import { ProductTour, type TourStep } from "@/components/ui/product-tour";

export function Demo() {
  const [open, setOpen] = useState(false);

  const steps: TourStep[] = [
    { title: "Welcome", body: "Let's take a quick tour around the interface." },
    { target: "#search-input", title: "Global Search", body: "Quickly locate projects and assets.", placement: "bottom" },
    { target: "#actions-panel", title: "Actions", body: "Create or import documents here.", placement: "left" },
  ];

  return (
    <div>
      <button onClick={() => setOpen(true)}>Start Tour</button>
      <ProductTour open={open} steps={steps} onClose={() => setOpen(false)} />
    </div>
  );
}
```

Declared props:
- `steps` (TourStep[]) — List of tour steps with target selector, title, body, and placement preference
- `open` (boolean) — Controls whether the spotlight tour overlay is open
- `onClose` ((reason: TourCloseReason) => void) — Callback triggered when tour completes, is skipped, or is dismissed
- `onStepChange` ((index: number) => void); default: undefined — Optional callback invoked whenever the active step index changes
- `startAt` (number); default: 0 — Zero-based initial step index when the tour opens
- `closeOnOverlayClick` (boolean); default: false — Whether clicking the dim overlay dismisses the tour
