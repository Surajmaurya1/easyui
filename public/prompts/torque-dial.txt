## Usage

Import path and example from component metadata:

```tsx
import { useState } from "react";
import { TorqueDial } from "@/components/ui/torque-dial";

export function Demo() {
  const [gain, setGain] = useState(48);

  return (
    <div className="p-8 flex items-center justify-center bg-[#090909] rounded-2xl border border-[#1F1F1F]">
      <TorqueDial
        value={gain}
        onChange={setGain}
        min={0}
        max={100}
        unit="%"
        label="Gain Level"
      />
    </div>
  );
}
```

Declared props:
- `value` (number) — Current numeric value
- `min` (number); default: 0 — Minimum dial threshold
- `max` (number); default: 100 — Maximum dial threshold
- `step` (number); default: 1 — Step resolution increment
- `onChange` ((val: number) => void) — Callback fired on rotation value updates
- `momentum` (boolean); default: true — Enables velocity-based inertial spin after drag release
- `size` (number); default: 120 — Diameter in pixels
- `unit` (string); default: '' — Optional unit suffix (e.g. "%", "dB", "°")
