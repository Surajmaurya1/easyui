## Usage

Import path and example from component metadata:

```tsx
import { useState } from "react";
import { VelocityToast } from "@/components/ui/velocity-toast";
import { PressButton } from "@/components/ui/press-button";

export function Demo() {
  const [open, setOpen] = useState(false);
  return (
    <>
      <PressButton onClick={() => setOpen(true)}>Show toast</PressButton>
      <VelocityToast
        open={open}
        onDismiss={() => setOpen(false)}
        title="File uploaded"
        description="easyui-2026-09.zip is ready."
        variant="success"
        position="bottom-right"
      />
    </>
  );
}
```

Declared props:
- `open` (boolean); default: undefined — Whether the toast is visible
- `onDismiss` (() => void); default: undefined — Called when toast is dismissed (auto or manual)
- `title` (string); default: undefined — Toast title
- `description` (string); default: undefined — Secondary description
- `duration` (number); default: 4000 — Auto-dismiss duration in ms
- `variant` ('default' | 'success' | 'warning' | 'error' | 'info'); default: 'default' — Visual tone
- `position` (VelocityToastPosition); default: 'bottom-center' — Screen placement
- `showProgress` (boolean); default: true — Show countdown progress bar
