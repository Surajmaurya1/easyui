## Usage

Import path and example from component metadata:

```tsx
import { useRef, useState } from "react";
import { WaveTransition } from "@/components/ui/WaveTransition";

export function Demo() {
  const [revealed, setRevealed] = useState(false);
  const triggerRef = useRef<HTMLButtonElement>(null);

  return (
    <WaveTransition
      revealed={revealed}
      revealedContent={<div className="grid h-64 place-items-center">New view</div>}
      source={triggerRef}
      palette="rainbow"
      overlay={(
        <button ref={triggerRef} onClick={() => setRevealed((value) => !value)}>
          Toggle view
        </button>
      )}
    >
      <div className="grid h-64 place-items-center">Current view</div>
    </WaveTransition>
  );
}
```

Declared props:
- `revealed` (boolean); default: required — Controls which view is shown and the wave direction
- `revealedContent` (React.ReactNode); default: required — The view revealed by the wave
- `children` (React.ReactNode); default: required — The starting view
- `type` ("vertical" | "horizontal" | "center"); default: "vertical" — Wave shape and default origin
- `source` (React.RefObject<HTMLElement | null>); default: undefined — Optional element whose center anchors the wave
- `palette` (WavePalette); default: "colorful" — Colorful, rainbow, ocean, sunset, green, mono, or three custom colors
- `duration` (number); default: 1.1 — Outgoing wave duration in seconds; return is 25% faster
- `overlay` (React.ReactNode); default: undefined — Persistent content layered above both views
- `onFinish` ((revealed: boolean) => void); default: undefined — Called when the wave finishes
